O0151 (OD TURN - STANDALONE - FANUC 0i-TD)
(JOHNFORD ST-130CH - DIAMETER PROGRAMMING)
(NO TEMPLATE REQUIRED - EDIT VALUES BELOW)
(BASED ON O0150 WITH VERIFIED RADIUS MATH)

(======= ENTER YOUR VALUES HERE =======)
#100=0.0     (TARGET OD DIAMETER)
#101=0.0     (START Z - USUALLY 0 AT FRONT FACE)
#102=0.0     (END Z - SHOULDER/FINAL Z - NEGATIVE)
#103=0.0     (STOCK DIAMETER - STARTING OD)
#104=0.1     (DOC - DEPTH OF CUT PER PASS RADIAL)
#105=0.02    (FINISH STOCK PER SIDE)
#106=0.010   (ROUGH FEED IPR)
#107=0.0     (END RADIUS R - OD TO SHOULDER FACE)
#108=0.0     (END CHAMFER B - ALTERNATIVE TO RADIUS)
#109=0.0     (FACE-UP DIA - 0 = NO FACE-UP)
#111=0.0     (RETRACT DIA - 0 = AUTO D+0.5)
#113=101     (ROUGH TOOL - T0101 = 101)
#114=202     (FINISH TOOL - T0202 = 202)
#115=600     (FINISH SFM)
#116=500     (ROUGH SFM)
#117=3000    (MAX RPM G50 CLAMP)
#118=2       (MODE 0=ROUGH 1=FINISH 2=BOTH)
(==========================================)

(=== VALIDATE INPUTS ===)
IF [#100 EQ 0] GOTO 990
IF [#103 EQ 0] GOTO 991
IF [#102 GE #101] GOTO 992
IF [#103 LE #100] GOTO 993
IF [#113 EQ 0] GOTO 994
IF [#114 EQ 0] GOTO 995

(DEFAULT RETRACT DIA)
IF [#111 NE 0] GOTO 1
#111=[#103+0.5]
N1

(VALIDATE RADIUS FITS IN Z LENGTH)
#120=[#101-#102]
IF [#107 LE #120] GOTO 2
#3000=10 (RADIUS LARGER THAN Z TRAVEL)
N2

(VALIDATE RADIUS FITS IN RADIAL STOCK)
(TARGET + 2R MUST BE <= STOCK DIA)
#121=[#100+[#107*2]]
IF [#121 LE #103] GOTO 3
#3000=11 (RADIUS EXCEEDS STOCK DIA)
N3

(VALIDATE FACE-UP DIA >= CORNER OUTER)
(S MUST BE AT LEAST TARGET+2R TO CLEAR CORNER)
IF [#109 EQ 0] GOTO 4
#122=[#100+[#107*2]]
IF [#108 GT #107] #122=[#100+[#108*2]]
IF [#109 GE #122] GOTO 4
#3000=12 (FACE-UP DIA BELOW CORNER)
N4

(=== INITIALIZE ===)
G20 G18 G40 G99 G97         (INCH, XZ PLANE, NO COMP, FEED/REV, RPM)
G28 U0 W0                    (HOME X THEN Z SAFELY)

(=== CALCULATE ROUGHING END Z ===)
(BASE: SHOULDER + FINISH STOCK)
#125=[#102+#105]

(CORNER START DIA - WHERE RADIUS/CHAM BEGINS)
(TARGET + 2R + 2C USED FOR SMART Z PULLBACK)
#126=[#100+[#107*2]+[#108*2]]

(=== ROUGHING ===)
IF [#118 EQ 1] GOTO 200     (SKIP IF FINISH ONLY)

(ROUGH TOOL CALL)
T#113
G50 S#117                    (MAX RPM CLAMP)
G96 S#116 M3                 (CSS + CW SPINDLE)
G0 X#111 Z[#101+0.1] M8     (APPROACH + COOLANT)

#130=#103                    (CURRENT X DIA TRACKER)

N100 (ROUGH LOOP)
#130=[#130-[#104*2]]         (STEP DOWN - 2X DOC FOR DIA)

(CLAMP TO FINISH ALLOWANCE)
IF [#130 GE [#100+#105]] GOTO 101
#130=[#100+#105]
N101

(--- SMART Z LIMIT: PULL BACK IF IN CORNER ZONE ---)
#131=#125                    (DEFAULT: FULL DEPTH)

(IF CURRENT DIA > CORNER START DIA, CUT FULL DEPTH)
IF [#130 GT #126] GOTO 105

(IN CORNER ZONE - PULL BACK BY R + SAFETY)
IF [#107 LE 0] GOTO 102
#131=[#125+#107+0.005]
N102
IF [#108 LE 0] GOTO 103
#131=[#125+#108+0.005]
N103

N105 (ROUGH PASS MOVES)
G0 X#111 Z[#101+0.05]
G0 X#130
G1 Z#131 F#106
G0 X#111
G0 Z[#101+0.05]

(LOOP CHECK)
IF [#130 GT [#100+#105]] GOTO 100

(ROUGH COMPLETE - RETRACT)
G0 X#111 Z[#101+0.5]

(EXIT IF ROUGH ONLY)
IF [#118 EQ 0] GOTO 300

(=== FINISH PASS ===)
N200
G28 U0 W0                    (HOME BEFORE TOOL CHANGE)
T#114
G50 S#117
G96 S#115 M3
G0 X#111 Z[#101+0.1] M8

(FINISH FEED = HALF ROUGH FEED)
#140=[#106/2]

(--- APPROACH AND ENTER AT FRONT ---)
G0 X[#100+0.05]
G1 X#100 Z#101 F#140

(--- CALCULATE CORNER START POINT ---)
(END Z OF OD TRAVEL = SHOULDER Z + R OR C, WHICHEVER LARGER)
#141=#102                    (DEFAULT = SHOULDER)
IF [#107 LE 0] GOTO 201
#141=[#102+#107]             (BACK OFF BY RADIUS)
N201
IF [#108 LE 0] GOTO 202
IF [#108 GT #107] #141=[#102+#108]
N202

(--- TRAVEL OD SURFACE TO CORNER START ---)
G1 Z#141 F#140

(--- CORNER TREATMENT ---)
IF [#108 GT 0] GOTO 210      (CHAMFER)
IF [#107 GT 0] GOTO 220      (RADIUS)
GOTO 230                      (SQUARE)

N210 (CHAMFER B)
(ENDPOINT: X = TARGET + 2*CHAMFER AT SHOULDER Z)
#142=[#100+[#108*2]]
G1 X#142 Z#102 F#140
GOTO 240

N220 (RADIUS R)
(=== RADIUS MATH VERIFICATION ===)
(EXAMPLE: TARGET=1.000  SHOULDER=-2.000  R=0.250)
(  ARC START  X=1.000  Z=-1.750  OD BACKED OFF R)
(  ARC END    X=1.500  Z=-2.000  TARGET+2R AT SHOULDER)
(  ARC CENTER X=1.500  Z=-1.750  TANGENT OD + FACE)
(  TANGENT AT START = ALONG Z  CONTINUOUS WITH OD)
(  TANGENT AT END   = ALONG X  READY FOR FACE-UP)
(  G2 CW CORRECT FOR OD END RADIUS FANUC LATHE)
(  R ENDS EXACTLY AT SHOULDER Z BEFORE FACING UP)
#143=[#100+[#107*2]]
G2 X#143 Z#102 R#107 F#140
GOTO 240

N230 (SQUARE CORNER)
G1 Z#102 F#140

(--- FACE-UP OR RETRACT ---)
N240
IF [#109 LE 0] GOTO 250
G1 X#109 F#140               (FACE UP ALONG SHOULDER TO S DIA)
GOTO 260

N250
(NO FACE-UP - DIRECT RETRACT)

N260
G0 X#111                     (RETRACT TO CLEARANCE DIA)
G0 Z[#101+1]                 (Z CLEAR)

(=== COMPLETE ===)
N300
G28 U0 W0 M9                 (HOME + COOLANT OFF)
M5                            (SPINDLE OFF)
M30                           (END OF PROGRAM)

(=== ALARM TARGETS ===)
N990

N991
#3000=2 (STOCK DIAMETER REQUIRED)
N992
#3000=3 (END Z MUST BE LESS THAN START Z)
N993
#3000=4 (STOCK DIA MUST BE GREATER THAN TARGET)
N994
#3000=5 (ROUGH TOOL STATION REQUIRED)
N995
#3000=6 (FINISH TOOL STATION REQUIRED)
